#include <bits/stdc++.h> 
int findSecondLargest(int n, vector<int> &arr)
{   
   int largestNo = arr[0];
   int secondLarge = -1;
    //First Largest:
   for(int i = 0; i < n; i++){
       if(arr[i] > largestNo){
           largestNo = arr[i];
       }
   }
   //Second largest:
   for(int i = 0; i < n; i++){
       if(arr[i] > secondLarge && arr[i] != largestNo){
           secondLarge = arr[i];
       }
   }

   return secondLarge;
}
